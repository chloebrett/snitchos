## System prompt

````
You write Stitch, a small statically-typed functional language. You have not
seen Stitch before — learn it from the reference, the standard library, and the
examples that follow.

Rules that are easy to get wrong:
- There are no loop keywords. Use recursion or the list combinators below.
- There is no if/else. Conditionals are `cond => a | b`, or a `match` block.
- Booleans are the words `and` / `or` / `not`. `&&` and `||` do not exist.
- There is no `return`. A block's value is its last expression.
- Exported items are prefixed `ext`; everything else is module-private.
- Comments explain *why*, never *what*. Keep them to one or two lines.
- Include `test "…" { expect … }` items covering the core logic.

Reply with exactly one fenced ```stitch block and nothing else.
````

# Stitch reference

// Modules & visibility
use List                                 // import a module
use text.{pad, wrap}                     // import named items
ext foo(...)                             // exported; without `ext`, private

// Exporting a type: `ext` goes BEFORE `prod`/`sum`, and every field that
// should be visible needs its own `ext` too.
ext prod Summary(ext count: Int, ext total: Int)
ext sum Shape = Circle(Int) | Square(Int)
// `ext` is NOT valid on `contract`, on an `on` block, or on `use`.

// Data — fields are immutable unless marked `mut`
prod Point(x: Int, y: Int)
prod Counter(mut n: Int)
sum Maybe<T>     = Some(T) | None        // built in
sum Result<T, E> = Ok(T)   | Err(E)      // built in

// Construction uses named fields
Point(x: 1, y: 2)

// Functions — expression body or block body
ext double(n: Int) -> Int = n * 2
ext describe(n: Int) -> Str = {
    let d = double(n)
    d > 10 => "big" | "small"
}

// Conditionals. `match` has two forms: guards, and patterns over a value.
cond => thenValue | elseValue
match { n == 0 => "zero"   n > 0 => "positive"   _ => "negative" }
match m { Some(v) => v   None => 0 }

// There is no `if`/`then`/`else` anywhere, including inside a lambda:
xs |> map(x -> x > 0 => "pos" | "neg")            // right
fold(xs, None, (acc, x) -> match acc { Some(_) => acc   None => Some(x) })

// Booleans are words, not symbols
a and b                                  // NOT &&
a or b                                   // NOT ||
n % d != 0

// Field access on a value is a plain dot. `@` is ONLY the receiver, and only
// inside an `on` block — never on a lambda parameter or a local.
b.start                                  // a field of `b`
@start                                   // field of the receiver, inside `on`

// Methods attach with `on`; `@` is the receiver
on Counter {
    bumped() -> Counter = Counter(n: @n + 1)
}
on Counter : Drawable {                  // `: Contract` declares conformance
    draw() uses Canvas = renderBar(@n)
}

// Capabilities are declared on the signature
report(xs: List<Int>) uses Telemetry = ...

// Short-circuit family
value?                                   // unwrap a Result/Maybe or short-circuit
user?.address                            // safe navigation

// Ranges and lists
[1, 2, 3]
2..n                                     // exclusive
1..=n                                    // inclusive

// Conversions are free functions, not methods
toStr(n)                                 // Int -> Str, NOT n.toString()
Str.parseInt(s)                          // Str -> Maybe<Int>

// Pipes and lambdas. A pipe supplies the FIRST argument, so these are identical:
fold(xs, 0, (acc, x) -> acc + x)
xs |> fold(0, (acc, x) -> acc + x)
xs |> map(x -> x * 2) |> filter(x -> x > 3)
xs |> map($.name)                        // `$a`/`$b` are positional placeholders

// Tests are ordinary items
test "double doubles" { expect double(2) == 4 }

# Standard library

Defined in Stitch itself on top of the built-ins above. Between the two lists,
this is everything that exists — but they are complete, so do not avoid a
function listed here.

// Stitch prelude — the part of the standard library written in Stitch itself,
// derived from the native core (fold/map/filter) + match + Maybe. Loaded into
// every program's globals before user code; user definitions shadow these.

// --- list combinators (derived from fold) ---

count(xs) = fold(xs, 0, (acc, _) -> acc + 1)

total(xs) = fold(xs, 0, (acc, x) -> acc + x)

any(xs, pred) = fold(xs, false, (acc, x) -> acc or pred(x))

all(xs, pred) = fold(xs, true, (acc, x) -> acc and pred(x))

contains(xs, value) = any(xs, x -> x == value)

// Note: written as a bare fold body, not `{ f(x)  () }` — `f(x) ()` would
// parse as a call of f(x)'s result (the maximal-munch corner).
each(xs, f) = fold(xs, (), (_, x) -> f(x))

find(xs, pred) =
    fold(xs, None, (acc, x) ->
        match acc {
            Some(_) => acc
            None => pred(x) => Some(x) | None
        })

// min/max fold to the extreme element, as a Maybe (None for an empty list).
// Comparable elements only (Int/Float/Str) — `<`/`>` are numeric+lexicographic.

min(xs) =
    fold(xs, None, (acc, x) ->
        match acc {
            None => Some(x)
            Some(m) => x < m => Some(x) | acc
        })

max(xs) =
    fold(xs, None, (acc, x) ->
        match acc {
            None => Some(x)
            Some(m) => x > m => Some(x) | acc
        })

// first/last as a Maybe (None for an empty list). Derived from fold, so for a
// List (eager); a Seq would diverge — force a prefix first.
first(xs) = fold(xs, None, (acc, x) -> match acc { None => Some(x)  Some(_) => acc })

last(xs) = fold(xs, None, (_, x) -> Some(x))

// flatten a List of Lists by concatenating — flatMap with the identity.
flatten(xss) = flatMap(xss, xs -> xs)

// --- the Try contract: `?` short-circuits on the failure case ---
// `x?` desugars to `x.isFailure() => return x | x.unwrap()`. Std implements it
// for the built-in Maybe/Result; a user type with a success/failure split can
// implement the same two methods to make `?` work on it.

contract Try {
    isFailure() -> Bool
    unwrap() -> @
}

on Maybe : Try {
    isFailure() = match @ { None => true  Some(_) => false }
    unwrap() = match @ { Some(v) => v  None => @ }
}

on Result : Try {
    isFailure() = match @ { Err(_) => true  Ok(_) => false }
    unwrap() = match @ { Ok(v) => v  Err(_) => @ }
}

// --- the Functor contract: `?.` maps a field access inside a container ---
// `x?.field` desugars to `x.map(v -> v.field)`. Unlike `?` (which unwraps), `map`
// re-wraps: it keeps the container shape, so the chain stays wrapped and
// short-circuits on the empty/failure case.

contract Functor {
    map(f) -> @
}

on Maybe : Functor {
    map(f) = match @ { Some(v) => Some(f(v))  None => None }
}

on Result : Functor {
    map(f) = match @ { Ok(v) => Ok(f(v))  Err(e) => Err(e) }
}

// --- Maybe helpers ---

isSome(m) = match m { Some(_) => true  None => false }

isNone(m) = match m { None => true  Some(_) => false }

unwrapOr(m, default) = match m { Some(v) => v  None => default }

mapMaybe(m, f) = match m { Some(v) => Some(f(v))  None => None }

andThen(m, f) = match m { Some(v) => f(v)  None => None }

// --- shell verbs ---

// `view f` — read file `f` and print its contents (or a not-found message). It
// runs with exactly `FsRead` + `ConsoleOut`: it may read files and write the
// console, nothing more — least authority, visible in the `uses` row.
view(f) uses FsRead, ConsoleOut = print(unwrapOr(readFile(f), "view: cannot read file"))

==== TEXT.ST ====

// text.st — the string utilities every other program here kept re-writing.
//
// Load with `:load lib/text.st`, or import the pieces you want:
// `use text.{pad, wrap, indent}`. Everything is pure — no capabilities, no
// telemetry — so these are safe to call from anywhere, including inside a
// `without Telemetry` block.

use Str
use List

// Pad `s` on the right with spaces until it is `width` wide. A string already
// at or past `width` comes back untouched, so this never truncates: a table
// column that overflows should look wrong, not lie.
ext pad(s: Str, width: Int) -> Str = {
    let short = width - Str.length(s)
    short <= 0 => s | s + spaces(short)
}

// Pad on the left instead — what numbers want, so a column of them lines up on
// the ones digit.
ext padLeft(s: Str, width: Int) -> Str = {
    let short = width - Str.length(s)
    short <= 0 => s | spaces(short) + s
}

test "pad widens to the column and never truncates" {
    expect pad("ok", 5) == "ok   "
    expect padLeft("7", 3) == "  7"
    // A value wider than its column must look wrong, not be cut.
    expect pad("overlong", 3) == "overlong"
}

// `n` spaces. Built by folding rather than recursing so a wide pad costs one
// pass and no stack.
ext spaces(n: Int) -> Str = repeatStr(" ", n)

// `s` repeated `n` times. `n <= 0` is the empty string, not an error — callers
// are computing widths, and a negative width means "nothing to add".
ext repeatStr(s: Str, n: Int) -> Str =
    fold(1..=n, "", (acc, _) -> acc + s)

test "repeat of a non-positive count is empty rather than an error" {
    expect repeatStr("ab", 0) == ""
    expect repeatStr("ab", -3) == ""
}

// Split `s` into lines of at most `width` characters, breaking at spaces.
// A word longer than `width` gets its own line rather than being cut in half —
// truncating a token is how log output starts lying about identifiers.
ext wrap(s: Str, width: Int) -> List<Str> =
    Str.split(s, " ")
        |> fold([], (lines, word) -> addWord(lines, word, width))
        |> reverse

// Prefix every line of `s` with `n` spaces. Blank lines stay blank — trailing
// whitespace on an empty line is invisible in a diff and noisy in a corpus.
ext indent(s: Str, n: Int) -> Str =
    Str.split(s, "\n")
        |> map(line -> line == "" => line | spaces(n) + line)
        |> Str.join("\n")

// Capitalise the first character, lowercase the rest: `"BOOT" -> "Boot"`.
ext titleCase(s: Str) -> Str = {
    Str.length(s) == 0 => s | {
        let head = Str.upper(Str.slice(s, 0, 1))
        let tail = Str.lower(Str.slice(s, 1, Str.length(s)))
        head + tail
    }
}

// One step of `wrap`: put `word` on the current line if it fits, else start a
// new one. `lines` is built in reverse (newest first) so appending is a cons.
addWord(lines: List<Str>, word: Str, width: Int) -> List<Str> = {
    let current = lines |> first |> unwrapOr("")
    let joined = current + " " + word
    match {
        count(lines) == 0 => [word]
        // `drop` is a Seq operation; a List drops its head by index.
        Str.length(joined) <= width => concat([joined], List.removeAt(lines, 0))
        _ => concat([word], lines)
    }
}

==== STATS.ST ====

// stats.st — summary statistics over a list of Ints.
//
// The shape `analyze` in `primes.st` reaches for by hand, extracted so a
// telemetry consumer can compute it once and pass it around as a value:
// `use stats.{summarise}` then `summarise(readings)`.
//
// Integer arithmetic throughout, and `mean` truncates. That is a deliberate
// choice, not an oversight: these numbers describe tick counts and byte
// totals, where a fractional mean is noise dressed as precision.

use List

// Everything worth knowing about a list of numbers, computed in one pass over
// a sorted copy. A `prod` rather than five return values so a caller can hold
// the summary, compare two of them, or emit it field by field.
ext prod Summary(
    ext count: Int,
    ext total: Int,
    ext mean: Int,
    ext median: Int,
    ext lowest: Int,
    ext highest: Int
)

// Summarise `xs`. An empty list summarises to all-zeroes rather than a
// `Maybe<Summary>`: every field of the empty summary is the true answer for
// no data (nothing summed to zero, no spread), and forcing every caller
// through an unwrap to learn that buys nothing.
ext summarise(xs: List<Int>) -> Summary = {
    let n = count(xs)
    n == 0 => empty() | {
        let sorted = sort(xs)
        // `sum` is the `sum`-type keyword, so the total needs another name.
        let added = total(xs)
        Summary(
            count: n,
            total: added,
            mean: added / n,
            median: middle(sorted, n),
            lowest: at(sorted, 0),
            highest: at(sorted, n - 1)
        )
    }
}

test "summarise reports the whole shape of a list" {
    let s = summarise([3, 1, 4, 1, 5])
    expect s.count == 5
    expect s.total == 14
    expect s.lowest == 1
    expect s.highest == 5
}

// Truncation is the documented contract, so it is asserted rather than left to
// be discovered: 14/5 is 2, not 3.
test "mean truncates rather than rounding" {
    expect summarise([3, 1, 4, 1, 5]).mean == 2
}

// The summary of no data.
ext empty() -> Summary =
    Summary(count: 0, total: 0, mean: 0, median: 0, lowest: 0, highest: 0)

// The empty list is the case a summariser gets wrong by dividing by zero or by
// forcing every caller through a `Maybe`. Zeroes are the true answer.
test "the empty list summarises to zeroes" {
    let s = summarise([])
    expect s.count == 0
    expect s.mean == 0
    expect range(s) == 0
}

// One reading has no spread, and its median, mean, and both extremes are all
// that reading — the degenerate case every branch has to agree on.
test "a single reading is its own summary" {
    let s = summarise([7])
    expect s.median == 7
    expect s.mean == 7
    expect range(s) == 0
}

// How far the data spreads: highest minus lowest. Zero for a single reading,
// which is correct — one sample has no spread — and the reason this is a
// function on `Summary` rather than a seventh field.
ext range(s: Summary) -> Int = s.highest - s.lowest

test "range is the distance between the extremes" {
    expect range(summarise([3, 1, 4, 1, 5])) == 4
}

// The median of an already-sorted list of `n` elements. Even lengths take the
// mean of the two middle values, so the median of `[1, 2, 3, 4]` is 2, not 2.5
// — the same truncation as `mean`, applied consistently.
middle(sorted: List<Int>, n: Int) -> Int = {
    let half = n / 2
    n % 2 == 1
        => at(sorted, half)
        | (at(sorted, half - 1) + at(sorted, half)) / 2
}

// The median has to sort first — reading the middle of the *input* returns
// whatever happened to be there, which is right often enough to hide the bug.
test "median sorts before it picks the middle" {
    expect summarise([5, 1, 3]).median == 3
    expect summarise([9, 1, 2, 8]).median == 5
}

// Index into a list, treating an out-of-range index as zero. Only reached with
// indices this module computed from the list's own length, so the default is
// unreachable in practice and present so the module has no partial functions.
at(xs: List<Int>, i: Int) -> Int = List.at(xs, i) |> unwrapOr(0)
